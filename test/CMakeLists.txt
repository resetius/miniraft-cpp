macro(ut name source lib)
  add_executable(${name} ${source})
  target_include_directories(${name} PRIVATE ${GTEST_INCLUDE_DIRS})
  target_link_directories(${name} PRIVATE ${GTEST_LIBRARY_DIRS})
  target_link_libraries(${name} PRIVATE ${lib} ${GTEST_LIBRARIES})

  add_test(NAME ${name} COMMAND ${name} --gtest_output=xml:${CMAKE_CURRENT_BINARY_DIR}/${name}.xml)
endmacro()
